import js import asyncio import time from pyscript import Element async def load_engine(): try: response = await js.fetch("lexical_engine.py") code = await response.text() exec(code, globals()) except Exception as exc: Element("finalOutput").write(f"Engine load failed: {exc}") asyncio.ensure_future(load_engine()) def run_lexical(): if "dfa_scan" not in globals(): Element("finalOutput").write("Engine loading... click again.") return code = Element("codeInput").value # DFA timings t0 = time.perf_counter() raw = dfa_scan(code) t1 = time.perf_counter() dfa_tokens = enrich(raw) t2 = time.perf_counter() # Regex timings r0 = time.perf_counter() raw_r = regex_scan(code) r1 = time.perf_counter() regex_tokens = enrich(raw_r) r2 = time.perf_counter() # Numbered output numbered = build_numbered_token_list(dfa_tokens) out = "" for i, (tok, cat) in enumerate(numbered, 1): out += f"{i}. {tok:<30} {cat}\n" Element("finalOutput").write(out) # DFA output Element("dfaOutput").write( "\n".join( f"{t.line}:{t.col} {t.type:<20} {t.value}" for t in dfa_tokens ) ) # Regex output Element("regexOutput").write( "\n".join( f"{t.line}:{t.col} {t.type:<20} {t.value}" for t in regex_tokens ) ) # Fundamental counts counts = { "KEYWORDS": 0, "IDENTIFIERS": 0, "NUMBERS": 0, "OPERATORS": 0, "STRINGS": 0, "COMMENTS": 0, "HEADERS": 0, "PUNCTUATION": 0 } for t in dfa_tokens: if t.type == "KEYWORD": counts["KEYWORDS"] += 1 elif t.type == "IDENTIFIER": counts["IDENTIFIERS"] += 1 elif "NUMBER" in t.type: counts["NUMBERS"] += 1 elif "OPERATOR" in t.type: counts["OPERATORS"] += 1 elif "STRING" in t.type: counts["STRINGS"] += 1 elif "COMMENT" in t.type: counts["COMMENTS"] += 1 elif t.type == "HEADER": counts["HEADERS"] += 1 elif t.type == "PUNCTUATION": counts["PUNCTUATION"] += 1 fund_text = "" for k, v in counts.items(): fund_text += f"{k:<15} : {v}\n" Element("fundOutput").write(fund_text) # Performance dfa_total = (t2 - t0) * 1000 regex_total = (r2 - r0) * 1000 perf = ( f"DFA Scan: {(t1-t0)*1000:.4f} ms\n" f"DFA Enrich: {(t2-t1)*1000:.4f} ms\n" f"DFA Total: {dfa_total:.4f} ms\n\n" f"Regex Scan: {(r1-r0)*1000:.4f} ms\n" f"Regex Enrich: {(r2-r1)*1000:.4f} ms\n" f"Regex Total: {regex_total:.4f} ms\n\n" ) if dfa_total < regex_total: perf += "✓ Result: DFA is faster" elif regex_total < dfa_total: perf += "✓ Result: Regex is faster" else: perf += "✓ Result: Both are equal in speed" Element("perfOutput").write(perf) # Expression stream expr_line = None for line in code.splitlines(): if "=" in line and ";" in line: expr_line = line.strip() break if expr_line: stream = token_stream_for_expression(expr_line) Element("exprOutput").write( "\n".join(f"{t:<20} -> {v}" for t, v in stream) ) else: Element("exprOutput").write("No expression found.")
